Practical No. 6(a) – Bayesian Linear Regression
# Aim
To implement Bayesian Linear Regression and explore the prior and posterior distributions using a synthetic dataset.

# Algorithm / Procedure
1. Import NumPy and Matplotlib libraries.
2. Generate a synthetic dataset containing input and output values.
3. Add a bias term to the input data.
4. Define the prior mean, prior covariance, and noise variance.
5. Calculate the posterior mean and covariance.
6. Generate samples from the prior and posterior distributions.
7. Plot both distributions and compare the results.

# Program

import numpy as np
import matplotlib.pyplot as plt

# Seed for reproducibility
np.random.seed(42)

# Create synthetic data
def generate_data(n_samples=10):
    X = np.linspace(0, 1, n_samples)
    y = 2 * X + 1 + 0.1 * np.random.randn(n_samples)
    return X.reshape(-1, 1), y

# Add bias term
def design_matrix(X):
    return np.hstack([np.ones((X.shape[0], 1)), X])

# Calculate posterior distribution
def compute_posterior(X, y, sigma2, prior_mean, prior_cov):
    XT_X = X.T @ X
    sigma_inv = np.linalg.inv(prior_cov)

    Sigma_n = np.linalg.inv(
        sigma_inv + XT_X / sigma2
    )

    mu_n = Sigma_n @ (
        sigma_inv @ prior_mean + X.T @ y / sigma2
    )

    return mu_n, Sigma_n

# Sample from distribution
def sample_posterior(mu, cov, n_samples=5):
    return np.random.multivariate_normal(mu, cov, n_samples)

# Generate data
X_raw, y = generate_data(n_samples=10)
X = design_matrix(X_raw)

# Define prior
prior_mean = np.zeros(X.shape[1])
prior_cov = np.eye(X.shape[1]) * 1.0

# Noise variance
sigma2 = 0.01

# Calculate posterior
mu_n, Sigma_n = compute_posterior(
    X, y, sigma2, prior_mean, prior_cov
)

# Plotting
x_plot = np.linspace(0, 1, 100).reshape(-1, 1)
X_plot = design_matrix(x_plot)

plt.figure(figsize=(12, 5))

# Plot prior samples
prior_samples = sample_posterior(
    prior_mean, prior_cov, n_samples=5
)

plt.subplot(1, 2, 1)

for w in prior_samples:
    y_plot = X_plot @ w
    plt.plot(x_plot, y_plot, lw=1)

plt.title("Samples from Prior")
plt.xlabel("x")
plt.ylabel("y")

# Plot posterior samples
posterior_samples = sample_posterior(
    mu_n, Sigma_n, n_samples=5
)

plt.subplot(1, 2, 2)

for w in posterior_samples:
    y_plot = X_plot @ w
    plt.plot(x_plot, y_plot, lw=1)

plt.scatter(X_raw, y, c="red", label="Data")
plt.title("Samples from Posterior")
plt.xlabel("x")
plt.ylabel("y")
plt.legend()

plt.tight_layout()
plt.show()


Result
Bayesian Linear Regression was successfully implemented.
The prior and posterior distributions were calculated and visualized using synthetic data.
